## Notes

A `HashSet` records visited coordinates. The second part alternates the active
coordinate pair by instruction index.

## Solution

```rust
use std::collections::HashSet;

fn part1(input: &str) -> usize {
    let (mut x, mut y) = (0, 0);
    let mut visited = HashSet::from([(0, 0)]);

    for instruction in input.chars() {
        match instruction {
            '^' => y += 1,
            'v' => y -= 1,
            '<' => x -= 1,
            '>' => x += 1,
            _ => unreachable!(),
        }
        visited.insert((x, y));
    }

    visited.len()
}

fn part2(input: &str) -> usize {
    let (mut first_x, mut first_y) = (0, 0);
    let (mut second_x, mut second_y) = (0, 0);
    let mut visited = HashSet::from([(0, 0)]);

    for (index, instruction) in input.chars().enumerate() {
        let (x, y) = if index % 2 == 0 {
            (&mut first_x, &mut first_y)
        } else {
            (&mut second_x, &mut second_y)
        };

        match instruction {
            '^' => *y += 1,
            'v' => *y -= 1,
            '<' => *x -= 1,
            '>' => *x += 1,
            _ => unreachable!(),
        }
        visited.insert((*x, *y));
    }

    visited.len()
}
```
