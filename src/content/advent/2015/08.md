## Notes

Scan string literals character by character instead of decoding into allocated
strings. Each escape pattern contributes a known length adjustment.

## Solution

```rust
use std::str::Chars;

fn count_chars(mut characters: Chars) -> usize {
    let mut count = 0;

    while let Some(character) = characters.next() {
        if character == '"' {
            continue;
        }

        if character == '\\' && matches!(characters.next(), Some('x')) {
            characters.next();
            characters.next();
        }

        count += 1;
    }

    count
}

fn part1(input: &str) -> usize {
    let (source_length, memory_length) = input.lines().fold((0, 0), |(source, memory), line| {
        (source + line.len(), memory + count_chars(line.chars()))
    });
    source_length - memory_length
}

fn count_encoded(mut encoded_length: usize, mut characters: Chars) -> usize {
    while let Some(character) = characters.next() {
        if character == '"' {
            encoded_length += 2;
        } else if character == '\\' {
            encoded_length += 2;
            if matches!(characters.next(), Some('x')) {
                characters.next();
                characters.next();
                encoded_length -= 1;
            }
        }
    }

    encoded_length
}

fn part2(input: &str) -> usize {
    let (source_length, encoded_length) = input.lines().fold((0, 0), |(source, encoded), line| {
        (source + line.len(), encoded + count_encoded(line.len(), line.chars()))
    });
    encoded_length - source_length
}
```
