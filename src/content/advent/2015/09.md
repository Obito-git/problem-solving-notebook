## Notes

Build an undirected distance map, enumerate every location order, and retain
the lowest or highest route length. This is practical for the small input size.

## Solution

```rust
use itertools::Itertools;
use std::collections::HashMap;

fn parse_distance(line: &str) -> (&str, &str, usize) {
    let (places, distance) = line.split_once(" = ").unwrap();
    let (from, to) = places.split_once(" to ").unwrap();
    (from, to, distance.parse().unwrap())
}

fn distances_from_input(input: &str) -> HashMap<&str, HashMap<&str, usize>> {
    let mut distances: HashMap<&str, HashMap<&str, usize>> = HashMap::new();

    for line in input.lines() {
        let (from, to, distance) = parse_distance(line);
        distances.entry(from).or_default().insert(to, distance);
        distances.entry(to).or_default().insert(from, distance);
    }

    distances
}

fn route_lengths(distances: &HashMap<&str, HashMap<&str, usize>>) -> impl Iterator<Item = usize> + '_ {
    distances.keys().permutations(distances.len()).map(|route| {
        route.windows(2)
            .map(|pair| distances[pair[0]][pair[1]])
            .sum()
    })
}

fn part1(input: &str) -> usize {
    route_lengths(&distances_from_input(input)).min().unwrap()
}

fn part2(input: &str) -> usize {
    route_lengths(&distances_from_input(input)).max().unwrap()
}
```
