## Notes

Treat the password as a mutable base-26 byte sequence. Increment until the
validation rules pass, skipping disallowed bytes during the increment.

## Solution

```rust
use itertools::Itertools;

fn matches_requirements(password: &[u8]) -> bool {
    let has_increasing_run = password
        .windows(3)
        .any(|window| window[0] + 1 == window[1] && window[1] + 1 == window[2]);
    let pair_count = password
        .windows(2)
        .filter(|window| window[0] == window[1])
        .unique()
        .count();

    has_increasing_run && pair_count >= 2
}

fn step(password: &mut [u8], position: usize) {
    match password[position] {
        b'z' => {
            password[position] = b'a';
            if position > 0 {
                step(password, position - 1);
            }
        }
        b'h' | b'k' | b'n' => password[position] += 2,
        _ => password[position] += 1,
    }
}

fn next_password(password: &mut [u8]) {
    loop {
        step(password, password.len() - 1);
        if matches_requirements(password) {
            return;
        }
    }
}

fn part1(mut input: Vec<u8>) -> String {
    next_password(&mut input);
    String::from_utf8(input).unwrap()
}

fn part2(mut input: Vec<u8>) -> String {
    next_password(&mut input);
    next_password(&mut input);
    String::from_utf8(input).unwrap()
}
```
