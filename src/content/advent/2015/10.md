## Notes

The sequence is stored as digits. `chunk_by` groups runs, and each run becomes
its count followed by its value.

## Solution

```rust
use itertools::Itertools;

fn next_look_and_say(input: Vec<u8>) -> Vec<u8> {
    let mut next = Vec::with_capacity(input.len() * 3 / 2);

    for (digit, group) in &input.iter().chunk_by(|&digit| digit) {
        next.push(group.count() as u8);
        next.push(*digit);
    }

    next
}

fn solve(input: Vec<u8>, iterations: usize) -> usize {
    let mut current = input;
    for _ in 0..iterations {
        current = next_look_and_say(current);
    }
    current.len()
}

fn part1(input: Vec<u8>) -> usize {
    solve(input, 40)
}

fn part2(input: Vec<u8>) -> usize {
    solve(input, 50)
}
```
