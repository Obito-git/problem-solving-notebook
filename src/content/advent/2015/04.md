## Notes

Brute-force candidate suffixes. `itoa` avoids allocating a decimal string for
each candidate, while the digest prefix is checked directly as bytes.

## Solution

```rust
fn part1(input: &str) -> usize {
    let mut buffer = itoa::Buffer::new();

    for candidate in 0.. {
        let mut context = md5::Context::new();
        context.consume(input.as_bytes());
        context.consume(buffer.format(candidate));
        let digest = context.finalize();

        if digest[0] == 0 && digest[1] == 0 && digest[2] & 0xF0 == 0 {
            return candidate;
        }
    }

    unreachable!()
}

fn part2(input: &str) -> usize {
    let mut buffer = itoa::Buffer::new();

    for candidate in 0.. {
        let mut context = md5::Context::new();
        context.consume(input.as_bytes());
        context.consume(buffer.format(candidate));
        let digest = context.finalize();

        if digest[0] == 0 && digest[1] == 0 && digest[2] == 0 {
            return candidate;
        }
    }

    unreachable!()
}
```
