## Notes

Both parts scan each line once. The second rule set keeps the first index of
each adjacent byte pair, which makes non-overlapping matches explicit.

## Solution

```rust
use std::collections::hash_map::Entry;
use std::collections::HashMap;

fn is_nice(line: &str) -> bool {
    let bytes = line.as_bytes();
    let mut vowel_count = 0;
    let mut has_double = false;

    for index in 0..line.len() {
        if let (Some(&left), Some(&right)) = (bytes.get(index), bytes.get(index + 1)) {
            match (left, right) {
                (b'a', b'b') | (b'c', b'd') | (b'p', b'q') | (b'x', b'y') => return false,
                _ if left == right => has_double = true,
                _ => {}
            }
        }

        if matches!(bytes[index] as char, 'a' | 'e' | 'i' | 'o' | 'u') {
            vowel_count += 1;
        }
    }

    vowel_count >= 3 && has_double
}

fn part1(input: &str) -> usize {
    input.lines().map(is_nice).filter(|is_nice| *is_nice).count()
}

fn is_nice2(line: &str) -> bool {
    let bytes = line.as_bytes();
    let mut first_pair_index: HashMap<(u8, u8), usize> = HashMap::new();
    let mut has_repeated_pair = false;
    let mut has_sandwiched_byte = false;

    for index in 0..line.len() {
        if let (Some(&left), Some(&right)) = (bytes.get(index), bytes.get(index + 1)) {
            match first_pair_index.entry((left, right)) {
                Entry::Occupied(entry) if index >= *entry.get() + 2 => has_repeated_pair = true,
                Entry::Vacant(entry) => {
                    entry.insert(index);
                }
                _ => {}
            }
        }

        if let (Some(&left), Some(&right)) = (bytes.get(index), bytes.get(index + 2))
            && left == right
        {
            has_sandwiched_byte = true;
        }

        if has_repeated_pair && has_sandwiched_byte {
            return true;
        }
    }

    false
}

fn part2(input: &str) -> usize {
    input.lines().map(is_nice2).filter(|is_nice| *is_nice).count()
}
```
