## Notes

This direct implementation materializes the grid. Commands are parsed into an
operation plus inclusive coordinate bounds, then applied cell by cell.

## Solution

```rust
use std::str::SplitWhitespace;

const TURN_ON: u8 = 1;
const TURN_OFF: u8 = 2;
const TOGGLE: u8 = 3;

fn parse_command(line: &str) -> (u8, (usize, usize), (usize, usize)) {
    let (command, rest) = if let Some(rest) = line.strip_prefix("turn on ") {
        (TURN_ON, rest)
    } else if let Some(rest) = line.strip_prefix("turn off ") {
        (TURN_OFF, rest)
    } else {
        (TOGGLE, line.strip_prefix("toggle ").unwrap())
    };

    let mut parts = rest.split_whitespace();
    let parse_coordinate = |parts: &mut SplitWhitespace| {
        let (x, y) = parts.next().unwrap().split_once(',').unwrap();
        (x.parse().unwrap(), y.parse().unwrap())
    };

    let start = parse_coordinate(&mut parts);
    parts.next();
    let end = parse_coordinate(&mut parts);
    (command, start, end)
}

fn part1(input: &str) -> i32 {
    let mut grid = vec![vec![false; 1000]; 1000];

    for line in input.lines() {
        let (command, (start_x, start_y), (end_x, end_y)) = parse_command(line);
        for x in start_x..=end_x {
            for y in start_y..=end_y {
                let cell = &mut grid[y][x];
                match command {
                    TURN_ON => *cell = true,
                    TURN_OFF => *cell = false,
                    TOGGLE => *cell = !*cell,
                    _ => unreachable!(),
                }
            }
        }
    }

    grid.into_iter()
        .flatten()
        .map(|is_on| if is_on { 1 } else { 0 })
        .sum()
}

fn part2(input: &str) -> usize {
    let mut grid = vec![vec![0usize; 1000]; 1000];

    for line in input.lines() {
        let (command, (start_x, start_y), (end_x, end_y)) = parse_command(line);
        for x in start_x..=end_x {
            for y in start_y..=end_y {
                let cell = &mut grid[y][x];
                match command {
                    TURN_ON => *cell += 1,
                    TURN_OFF => *cell = cell.saturating_sub(1),
                    TOGGLE => *cell += 2,
                    _ => unreachable!(),
                }
            }
        }
    }

    grid.into_iter().flatten().sum()
}
```
