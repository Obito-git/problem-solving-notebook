## Notes

Parse each output wire into an expression. Recursive resolution with a cache
turns repeated wire lookups into constant-time reads after the first evaluation.

## Solution

```rust
use std::collections::HashMap;

#[derive(Debug, Hash, PartialEq, Eq, Copy, Clone)]
struct WireIdentifier(u8, u8);

impl From<&str> for WireIdentifier {
    fn from(value: &str) -> Self {
        let raw = value.as_bytes();
        WireIdentifier(raw[0], raw.get(1).copied().unwrap_or(0))
    }
}

#[derive(Debug, Clone, Copy)]
enum Operand {
    Wire(WireIdentifier),
    Value(u16),
}

impl From<&str> for Operand {
    fn from(value: &str) -> Self {
        if let Ok(value) = value.parse::<u16>() {
            Operand::Value(value)
        } else {
            Operand::Wire(WireIdentifier::from(value))
        }
    }
}

enum Gate {
    Assign(Operand),
    And(Operand, Operand),
    Or(Operand, Operand),
    Lshift(Operand, u16),
    Rshift(Operand, u16),
    Not(Operand),
}

impl From<&str> for Gate {
    fn from(value: &str) -> Self {
        let mut tokens = value.split_whitespace();
        match tokens.clone().count() {
            1 => Gate::Assign(Operand::from(tokens.next().unwrap())),
            2 => {
                if tokens.next() != Some("NOT") {
                    panic!("invalid unary gate");
                }
                Gate::Not(Operand::from(tokens.next().unwrap()))
            }
            3 => {
                let left = Operand::from(tokens.next().unwrap());
                let operator = tokens.next().unwrap();
                let right = tokens.next().unwrap();

                match operator {
                    "AND" => Gate::And(left, Operand::from(right)),
                    "OR" => Gate::Or(left, Operand::from(right)),
                    "LSHIFT" => Gate::Lshift(left, right.parse().unwrap()),
                    "RSHIFT" => Gate::Rshift(left, right.parse().unwrap()),
                    _ => panic!("invalid operator"),
                }
            }
            _ => panic!("invalid gate"),
        }
    }
}

fn parse_line(line: &str) -> (WireIdentifier, Gate) {
    let (gate, wire) = line.split_once(" -> ").unwrap();
    (WireIdentifier::from(wire), Gate::from(gate))
}

fn resolve_operand(
    gates: &HashMap<WireIdentifier, Gate>,
    values: &mut HashMap<WireIdentifier, u16>,
    operand: &Operand,
) -> u16 {
    match operand {
        Operand::Value(value) => *value,
        Operand::Wire(wire) => find_signal(gates, values, wire),
    }
}

fn find_signal(
    gates: &HashMap<WireIdentifier, Gate>,
    values: &mut HashMap<WireIdentifier, u16>,
    wire: &WireIdentifier,
) -> u16 {
    if let Some(&value) = values.get(wire) {
        return value;
    }

    let value = match gates.get(wire).unwrap() {
        Gate::Assign(operand) => resolve_operand(gates, values, operand),
        Gate::And(left, right) => {
            resolve_operand(gates, values, left) & resolve_operand(gates, values, right)
        }
        Gate::Or(left, right) => {
            resolve_operand(gates, values, left) | resolve_operand(gates, values, right)
        }
        Gate::Lshift(operand, shift) => resolve_operand(gates, values, operand) << shift,
        Gate::Rshift(operand, shift) => resolve_operand(gates, values, operand) >> shift,
        Gate::Not(operand) => !resolve_operand(gates, values, operand),
    };

    values.insert(*wire, value);
    value
}

fn part1(input: &str, wire: WireIdentifier) -> u16 {
    let gates = input.lines().map(parse_line).collect();
    find_signal(&gates, &mut HashMap::new(), &wire)
}

fn part2(input: &str, wire: WireIdentifier) -> u16 {
    let gates = input.lines().map(parse_line).collect();
    let mut values = HashMap::new();
    let first = find_signal(&gates, &mut values, &wire);
    values.clear();
    values.insert(WireIdentifier(b'b', 0), first);
    find_signal(&gates, &mut values, &wire)
}
```
