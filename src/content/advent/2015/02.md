## Notes

Parse each record once through an iterator. The second calculation sorts the
three dimensions to select the two smallest values.

## Solution

```rust
fn dimensions_from_input(input: &str) -> impl Iterator<Item = (u32, u32, u32)> + use<'_> {
    input.lines().map(|line| {
        let mut dimensions = line.split('x');
        let length = dimensions.next().unwrap().parse().unwrap();
        let width = dimensions.next().unwrap().parse().unwrap();
        let height = dimensions.next().unwrap().parse().unwrap();
        (length, width, height)
    })
}

fn part1(input: &str) -> u32 {
    dimensions_from_input(input).fold(0, |total, (length, width, height)| {
        let first = length * width;
        let second = width * height;
        let third = height * length;
        total + 2 * first + 2 * second + 2 * third + first.min(second).min(third)
    })
}

fn part2(input: &str) -> u32 {
    dimensions_from_input(input).fold(0, |total, (length, width, height)| {
        let mut dimensions = [length, width, height];
        dimensions.sort();
        total + length * width * height + 2 * dimensions[0] + 2 * dimensions[1]
    })
}
```
