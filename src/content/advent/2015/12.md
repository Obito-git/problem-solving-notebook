## Notes

Part one extracts signed integers directly from the character stream. Part two
walks bytes with a stack of object sums and discards a completed object when it
contains the excluded value.

## Solution

```rust
fn can_be_number(character: char) -> bool {
    character.is_numeric() || character == '-'
}

fn part1(input: &str) -> i32 {
    let mut sum = 0;
    let mut characters = input.chars().peekable();

    while characters.peek().is_some() {
        characters.by_ref().skip_while(|character| !can_be_number(*character));
        let number: String = characters.by_ref().take_while(|character| can_be_number(*character)).collect();
        if !number.is_empty() {
            sum += number.parse::<i32>().unwrap();
        }
    }

    sum
}

fn byte_can_be_number(byte: u8) -> bool {
    (byte as char).is_numeric() || byte == b'-'
}

fn part2(input: &[u8]) -> i32 {
    let mut stack: Vec<(i32, bool)> = vec![(0, false)];
    let mut index = 0;

    while index < input.len() {
        match input[index] {
            b'{' => {
                stack.push((0, false));
                index += 1;
            }
            b'}' => {
                let (sum, has_excluded_value) = stack.pop().unwrap();
                if !has_excluded_value {
                    stack.last_mut().unwrap().0 += sum;
                }
                index += 1;
            }
            b':' => {
                index += 1;
                while input.get(index).is_some_and(u8::is_ascii_whitespace) {
                    index += 1;
                }
                if input[index..].starts_with(b"\"red\"") {
                    stack.last_mut().unwrap().1 = true;
                    index += 5;
                }
            }
            byte if byte_can_be_number(byte) => {
                let start = index;
                while input.get(index).is_some_and(|byte| byte_can_be_number(*byte)) {
                    index += 1;
                }
                let number = std::str::from_utf8(&input[start..index]).unwrap().parse::<i32>().unwrap();
                stack.last_mut().unwrap().0 += number;
            }
            _ => index += 1,
        }
    }

    stack[0].0
}
```
